        // 4. Toggle Chart View (Split vs Combined)
        function switchChartView(view) {
            const splitContainer = document.getElementById('split-charts-container');
            const combinedContainer = document.getElementById('combined-chart-container');
            const btnSplit = document.getElementById('btn-split-view');
            const btnCombined = document.getElementById('btn-combined-view');

            if (view === 'split') {
                splitContainer.classList.remove('hidden');
                combinedContainer.classList.add('hidden');

                btnSplit.classList.add('bg-white', 'text-slate-900', 'shadow-sm');
                btnSplit.classList.remove('text-slate-600');

                btnCombined.classList.remove('bg-white', 'text-slate-900', 'shadow-sm');
                btnCombined.classList.add('text-slate-600');
            } else {
                splitContainer.classList.add('hidden');
                combinedContainer.classList.remove('hidden');

                btnCombined.classList.add('bg-white', 'text-slate-900', 'shadow-sm');
                btnCombined.classList.remove('text-slate-600');

                btnSplit.classList.remove('bg-white', 'text-slate-900', 'shadow-sm');
                btnSplit.classList.add('text-slate-600');
            }
        }
